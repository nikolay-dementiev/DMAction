// an include do not apply there. The force_unwrapping rule has none, so no line here forces an
// unwrap. No line expects a diagnostic.

    preconditionFailure("x")
    assertionFailure("x")
    precondition(false)
    assert(false)
    dispatchPrecondition(condition: .onQueue(.main))
    abort()
    exit(1)
}
